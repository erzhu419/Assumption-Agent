                if state[j] == 0 and v * (t - 0.5 * diag[j] * v) < l0:
                    v = 0.
                step = v - w[j]
                if step:
                    g -= step * H[:, j]
                    w[j] = v
                    delta = max(delta, abs(step))
            if delta <= 1e-10:
                break
        val = score(w)
        if val < upper:
            upper, best = val, w

    def relax(w, state):

        w[state == -1] = 0.

                if step:

                    w[j] = v
                    delta = max(delta, abs(step))
            if delta <= 1e-10:

        res = y - x @ w
        gradient = x.T @ res
        z = abs(gradient + d * w)
        v = np.minimum(z / a, box)
        on = l0 - v * (z - 0.5 * a * v)

        val += on[state == 1].sum() + np.minimum(on[state == 0], 0.).sum()
        return float(val - sign_guard - threshold_guard), on

                       (1 - ratio) * (l0 - 0.5 * a[free] * knee[free]**2 * ratio), 0.)

            'certified_lower_bound': min(leaf['lb'] for leaf in leaves),
            'subcube_leaves': leaves}
