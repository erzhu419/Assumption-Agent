"""Finite-volume Helmholtz source inversion from polar receiver data."""

import numpy as np
from scipy.sparse import coo_matrix
from scipy.sparse.linalg import splu


def _responses(problem):
    """Solve for the complex pressure generated by each unit source."""
    n_sq = np.asarray(problem['n_sq'])
    nx, ny = n_sq.shape
    hx, hy = problem['Lx'] / nx, problem['Ly'] / ny
    k0 = problem['k0']
    cells = np.arange(nx * ny).reshape(nx, ny)
    diagonal = (-hx * hy * k0**2 * n_sq).astype(np.complex128)
    impedance = -1j * k0 * problem['beta_imp']
    diagonal[0, :] += hy * impedance
    diagonal[-1, :] += hy * impedance
    diagonal[:, 0] += hx * impedance
    diagonal[:, -1] += hx * impedance

    rows, cols, values = [], [], []
    for left, right, weight in ((cells[:-1, :], cells[1:, :], hy / hx),
                                (cells[:, :-1], cells[:, 1:], hx / hy)):
        left, right = left.ravel(), right.ravel()
        np.add.at(diagonal.ravel(), left, weight)
        np.add.at(diagonal.ravel(), right, weight)
        rows.extend((left, right))
        cols.extend((right, left))
        values.extend((np.full(left.size, -weight), np.full(right.size, -weight)))
    rows.append(cells.ravel())
    cols.append(cells.ravel())
    values.append(diagonal.ravel())
    operator = coo_matrix((np.concatenate(values),
                           (np.concatenate(rows), np.concatenate(cols))),
                          shape=(cells.size, cells.size)).tocsc()
    rhs = hx * hy * np.asarray(problem['basis'], dtype=np.complex128).reshape(-1, cells.size).T
    return splu(operator).solve(rhs).reshape(nx, ny, -1)


def _robust_real_fit(design, observed, alpha):
    """Trim gross errors by receiver, then refit the noise-consistent receivers."""
    n, m = design.shape
    penalty = np.sqrt(alpha) * np.eye(m)

    def fit(indices, regularized=False):
        block = design[indices]
        target = observed[indices]
        x = np.concatenate((block.real, block.imag), axis=0)
        y = np.concatenate((target.real, target.imag))
        if regularized and alpha > 0:
            x = np.vstack((x, penalty))
            y = np.concatenate((y, np.zeros(m)))
        return np.linalg.lstsq(x, y, rcond=None)[0]

    def residual(a):
        return np.abs(design @ a - observed)

    # A fixed seed makes the small set of starting subsets reproducible.
    # Select by trimmed error, so one impulsive channel cannot win a restart.
    keep = min(n, max(m, int(np.ceil(0.85 * n))))
    rng = np.random.default_rng(0)
    starts = [np.arange(n)]
    starts.extend(rng.choice(n, size=min(n, 2 * m), replace=False) for _ in range(20))
    best_cost = np.inf
    for start in starts:
        a = fit(start)
        previous = None
        for _ in range(10):
            indices = np.sort(np.argpartition(residual(a), keep - 1)[:keep])
            if np.array_equal(indices, previous):
                break
            a = fit(indices)
            previous = indices
        trimmed = np.partition(residual(a), keep - 1)[:keep]
        cost = trimmed @ trimmed
        if cost < best_cost:
            best_cost, best = cost, a

    errors = residual(best)
    # For circular complex Gaussian noise, |residual| is Rayleigh-distributed.
    sigma = np.median(errors) / np.sqrt(2 * np.log(2))
    roundoff = 64 * np.finfo(float).eps * np.max(np.abs(observed))
    indices = np.flatnonzero(errors <= max(3.8 * sigma, roundoff))
    if indices.size < keep:
        indices = np.argpartition(errors, keep - 1)[:keep]
    return fit(indices, regularized=True)


def invert_helmholtz_sources_and_phase(problem: dict) -> dict:
    """Recover real source amplitudes, oscillator phase, and pressure magnitude."""
    pilot = problem['pilot_calibration']
    known = np.asarray(pilot['known_field_re']) + 1j * np.asarray(pilot['known_field_im'])
    pilot_observed = pilot['amplitude_obs'] * np.exp(1j * np.asarray(pilot['wrapped_phase_obs']))
    phase = float(np.angle(np.vdot(known, pilot_observed)))
    if phase <= -np.pi:
        phase = float(np.pi)

    response = _responses(problem)
    survey = problem['survey']
    i, j = np.asarray(survey['sensor_ij']).T
    observed = survey['amplitude_obs'] * np.exp(1j * (np.asarray(survey['wrapped_phase_obs']) - phase))
    amplitudes = _robust_real_fit(response[i, j], observed, survey['tikhonov_alpha'])
    return {
        'source_amplitudes': amplitudes,
        'global_phase_rad': phase,
        'predicted_field_abs': np.abs(response @ amplitudes),
    }
